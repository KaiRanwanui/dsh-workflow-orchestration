return {
  inject: ['timer'],
  apply(ctx) {
    const slots = ctx.get('slots')
    if (!slots) return

    let latest = null
    let lastError = null
    const listeners = new Set()
    let pollingActive = false

    function fingerprint(state) {
      if (!state || !state.tasks) return ''
      const t = state.tasks.map((x) => x.id + ':' + x.status).join(',')
      return [state.stage || '', state.gateResult || '', state.retries || 0, t].join('|')
    }

    function publish(state) {
      const fp = state ? fingerprint(state) : ''
      if (fp && latest && latest.__fp === fp) return
      if (state) state.__fp = fp
      latest = state
      lastError = state ? null : (lastError || 'disconnected')
      listeners.forEach((fn) => { try { fn() } catch (e) {} })
    }

    ctx.effect(() => {
      if (pollingActive) return
      pollingActive = true
      let stop = false
      let currentRoot = null

      window.__wfSetRoot = function (root) {
        currentRoot = root
      }

      const refresh = async () => {
        if (stop) return
        try {
          const s = await host.call('wf:status', { workspaceRoot: currentRoot || '' })
          if (!stop) publish(s)
        } catch (e) {
          if (!stop) { lastError = e && e.message ? e.message : String(e); listeners.forEach((fn) => { try { fn() } catch (e2) {} }) }
        }
      }

      refresh()
      const d = ctx.interval(refresh, 2000)
      return () => { stop = true; pollingActive = false; d(); delete window.__wfSetRoot }
    })

    const NODE_COLORS = {
      PENDING: '#9ca3af',
      RUNNING: '#3b82f6',
      DONE: '#22c55e',
      FAILED: '#ef4444',
      SKIPPED: '#f59e0b',
    }
    const STAGE_LABELS = {
      PENDING: '待启动',
      RUNNING: '运行中',
      COMPLETED: '已完成',
      FAILED: '已失败',
    }

    const DagCanvas = React.memo(function DagCanvas(props) {
      const { stage, gateResult, tasks, selectedId, onSelect, workflowName, retries, error } = props
      const mainColor = (() => {
        if (stage === 'COMPLETED') return gateResult === 'FAIL' ? NODE_COLORS.FAILED : NODE_COLORS.DONE
        if (stage === 'FAILED') return NODE_COLORS.FAILED
        if (stage === 'RUNNING') return NODE_COLORS.RUNNING
        return NODE_COLORS.PENDING
      })()

const gw = 132, gh = 46, gap = 44
      const leftPad = 24, cy = 28, groupLabelH = 20
      const items = Array.isArray(tasks) ? tasks : []
      const svgChildren = []
      let x = leftPad
      const arrowId = 'wf-dash-arrow'

      // Group consecutive loop iterations
      const groups = []
      for (let gi = 0; gi < items.length;) {
        const t = items[gi]
        if (t._loopGroup) {
          const g = { key: t._loopGroup, name: t._loopGroupName || t._loopGroup, items: [] }
          while (gi < items.length && items[gi]._loopGroup === g.key) { g.items.push(items[gi]); gi++ }
          groups.push(g)
        } else {
          groups.push({ key: null, name: null, items: [items[gi]] })
          gi++
        }
      }

      // Background layer: group rects
      x = leftPad
      for (const g of groups) {
        const groupW = g.items.length * (gw + gap) - gap
        if (g.key) {
          const bgPadding = 8
          svgChildren.push(
            React.createElement('rect', {
              key: 'bg' + g.key, x: x - bgPadding / 2, y: cy - 6 - groupLabelH,
              width: groupW + bgPadding, height: gh + bgPadding + groupLabelH,
              rx: 10, fill: 'rgba(148,163,184,0.07)',
            }),
            React.createElement('text', {
              key: 'gl' + g.key, x: x + groupW / 2, y: cy - 6 - groupLabelH + 14,
              textAnchor: 'middle', fill: '#94a3b8', fontSize: 11, fontWeight: 600,
            }, '\u21BB ' + g.name),
          )
        }
        x += g.items.length * (gw + gap)
      }

      // Node layer: iterate groups
      x = leftPad
      for (const g of groups) {
        g.items.forEach((t, j) => {
          const isSel = selectedId === t.id
          const fill = NODE_COLORS[t.status] || NODE_COLORS.PENDING

          svgChildren.push(
            React.createElement('rect', {
              key: 'r' + t.id, x, y: cy, width: gw, height: gh, rx: 8,
              fill, opacity: 0.92,
              stroke: isSel ? '#fff' : 'transparent', strokeWidth: isSel ? 3 : 0,
              cursor: 'pointer',
              onClick: () => onSelect(t.id),
            }),
            React.createElement('text', {
              key: 't' + t.id, x: x + gw / 2, y: cy + gh / 2 - 2,
              textAnchor: 'middle', fill: '#fff', fontSize: 12, fontWeight: 600,
            }, t.name || t.id),
            React.createElement('text', {
              key: 'u' + t.id, x: x + gw / 2, y: cy + gh / 2 + 14,
              textAnchor: 'middle', fill: 'rgba(255,255,255,0.85)', fontSize: 10,
            }, g.key ? ('#' + t._loopItem) : t.id),
          )

          if (j < g.items.length - 1) {
            svgChildren.push(
              React.createElement('line', {
                key: 'l' + t.id,
                x1: x + gw, y1: cy + gh / 2,
                x2: x + gw + gap, y2: cy + gh / 2,
                stroke: '#94a3b8', strokeWidth: 2,
                markerEnd: 'url(#' + arrowId + ')',
              }),
            )
          }
          x += gw + gap
        })
      }

      svgChildren.push(
        React.createElement('defs', { key: 'defs' },
          React.createElement('marker', {
            id: arrowId, viewBox: '0 0 10 10', refX: 9, refY: 5,
            markerWidth: 7, markerHeight: 7, orient: 'auto-start-reverse',
          },
            React.createElement('path', { d: 'M 0 0 L 10 5 L 0 10 z', fill: '#94a3b8' }),
          ),
        ),
      )

      const svgWidth = leftPad * 2 + Math.max(groups.reduce((s, g) => s + g.items.length, 0), 1) * (gw + gap)
      const svgHeight = cy * 2 + gh + groupLabelH

      return React.createElement('div', {
        style: {
          padding: '14px 18px 10px',
          borderBottom: '1px solid rgba(148,163,184,0.3)',
        },
      }, [
        React.createElement('div', {
          key: 'statusBar',
          style: {
            display: 'flex', alignItems: 'center', gap: 14,
            marginBottom: 8, fontSize: 12, color: '#64748b',
            flexWrap: 'wrap',
          },
        }, [
          React.createElement('span', { key: 'w', style: { fontWeight: 600, color: '#334155' } }, workflowName || '(无工作流)'),
          React.createElement('span', { key: 's' },
            '阶段: ',
            React.createElement('span', { style: { color: mainColor, fontWeight: 700 } },
              (stage || '—') + ' ' + (STAGE_LABELS[stage] || ''),
            ),
          ),
          React.createElement('span', { key: 'g' },
            '门禁: ',
            gateResult
              ? React.createElement('span', {
                  style: { color: gateResult === 'PASS' ? NODE_COLORS.DONE : NODE_COLORS.FAILED, fontWeight: 700 },
                }, gateResult)
              : '—',
          ),
          retries > 0 ? React.createElement('span', { key: 'r' }, '重试: ', retries) : null,
          error ? React.createElement('span', { key: 'e', style: { color: NODE_COLORS.FAILED } }, error) : null,
        ]),
        React.createElement('svg', {
          key: 'svg',
          width: svgWidth,
          height: svgHeight,
          xmlns: 'http://www.w3.org/2000/svg',
          style: {
            background: 'rgba(148,163,184,0.08)',
            borderRadius: 8,
            width: '100%',
            maxWidth: svgWidth,
          },
        }, svgChildren),
      ])
    })

    slots.inject('conversation.view', () => {
      return slots.register(
        {
          name: 'conversation.view',
          id: 'workflow',
          order: 25,
          label: () => '工作流',
        },
        function WorkflowViewFactory(props) {
          const workspaceHook = props.useWorkspaces

          function WorkflowView() {
            const [, forceUpdate] = React.useReducer((x) => x + 1, 0)
            const [selectedId, setSelectedId] = React.useState(null)
            const [skillText, setSkillText] = React.useState(null)
            const [skillLoading, setSkillLoading] = React.useState(false)
            const [workspaceRoot, setWorkspaceRoot] = React.useState('')
            const [configLoaded, setConfigLoaded] = React.useState(false)

            React.useEffect(() => {
              listeners.add(forceUpdate)
              return () => listeners.delete(forceUpdate)
            }, [])

            React.useEffect(() => {
              if (configLoaded) return
              try {
                if (workspaceHook) {
                  const wsList = workspaceHook()
                  if (wsList && wsList.data && Array.isArray(wsList.data.items) && wsList.data.items.length > 0) {
                    const p = wsList.data.items[0].path
                    if (p) {
                      const root = String(p).replace(/\\/g, '/')
                      setWorkspaceRoot(root)
                      if (typeof window.__wfSetRoot === 'function') window.__wfSetRoot(root)
                      setConfigLoaded(true)
                      return
                    }
                  }
                }
              } catch (e) {}
              host.call('wf:config', { workspaceRoot: 'C:/Users/ranwa/dsh_workspace' }).then((r) => {
                if (r && r.valid) {
                  setWorkspaceRoot(r.workspaceRoot)
                  if (typeof window.__wfSetRoot === 'function') window.__wfSetRoot(r.workspaceRoot)
                }
                setConfigLoaded(true)
              }).catch(() => setConfigLoaded(true))
            }, [workspaceHook, configLoaded])

            React.useEffect(() => {
              if (!selectedId || !workspaceRoot) return
              const snap = latest
              const tasks = (snap && snap.state && Array.isArray(snap.state.tasks)) ? snap.state.tasks : []
              const task = tasks.find((t) => t.id === selectedId)
              const path = task && task.processor ? task.processor : null
              if (!path) {
                setSkillText('（该节点未配置处理器技能文件）')
                setSkillLoading(false)
                return
              }
              let cancelled = false
              setSkillLoading(true)
              host.call('wf:skill', { path }).then((r) => {
                if (!cancelled) {
                  setSkillText(r && r.text ? r.text : '（技能文件为空）')
                  setSkillLoading(false)
                }
              }).catch((e) => {
                if (!cancelled) { setSkillText('（加载失败: ' + (e.message || String(e)) + '）'); setSkillLoading(false) }
              })
              return () => { cancelled = true }
            }, [selectedId, workspaceRoot])

            const onSelect = React.useCallback((id) => {
              setSelectedId((prev) => prev === id ? null : id)
              setSkillText(null)
              setSkillLoading(false)
            }, [])

            const snap = latest
            const stateData = (snap && snap.state) ? snap.state : null
            const tasks = stateData && Array.isArray(stateData.tasks) ? stateData.tasks : []
            const hasData = stateData && stateData.workflow

            if (!configLoaded) {
              return React.createElement('div', {
                style: {
                  display: 'flex', alignItems: 'center', justifyContent: 'center',
                  height: '100%', minHeight: 420, color: '#9ca3af', fontSize: 13,
                },
              }, '正在连接工作流引擎…')
            }

            if (!workspaceRoot) {
              return React.createElement('div', {
                style: {
                  display: 'flex', alignItems: 'center', justifyContent: 'center',
                  height: '100%', minHeight: 420, color: '#9ca3af', fontSize: 13,
                  flexDirection: 'column', gap: 8,
                },
              }, [
                React.createElement('span', { key: 't1' }, '未检测到工作区'),
                React.createElement('span', { key: 't2', style: { fontSize: 11 } }, '请在 DSH 中打开一个工作区，然后运行 Workflow Orchestrator 预设'),
              ])
            }

            if (!hasData) {
              const hint = snap && snap.error
                ? '工作流数据暂不可用: ' + snap.error
                : '工作流仪表盘连接中…'
              return React.createElement('div', {
                style: {
                  display: 'flex', alignItems: 'center', justifyContent: 'center',
                  height: '100%', minHeight: 420, color: '#9ca3af', fontSize: 13,
                  border: '1px dashed rgba(148,163,184,0.35)', borderRadius: 8,
                  margin: 12, background: 'rgba(148,163,184,0.05)',
                },
              }, hint)
            }

            const selectedTask = selectedId ? tasks.find((t) => t.id === selectedId) : null
            let detailTitle = '点击上方节点查看详情'
            let detailContent = null

            if (selectedTask) {
              detailTitle = selectedTask.name + ' (' + selectedTask.id + ')'
              const detailLines = [
                '类型: ' + (selectedTask.type || 'llm-task'),
                '状态: ' + (selectedTask.status || '—'),
                '重试: ' + (selectedTask.retries || 0),
              ]
              if (selectedTask.gateChecker) {
                detailLines.push('门禁: ' + selectedTask.gateChecker.split('/').pop())
              }
              if (selectedTask.gateResult) {
                detailLines.push('门禁结果: ' + selectedTask.gateResult)
              }
              if (selectedTask.processor) {
                detailLines.push('处理器: ' + selectedTask.processor)
              }

              detailContent = React.createElement('div', { key: 'info', style: { fontSize: 12, lineHeight: 1.8, color: '#475569' } },
                detailLines.map((line, i) => React.createElement('div', { key: i }, line)),
              )

              if (skillLoading) {
                detailContent = React.createElement('div', { key: 'loading', style: { fontSize: 12, color: '#9ca3af' } }, '技能加载中…')
              } else if (skillText) {
                detailContent = React.createElement('div', null, [
                  detailContent,
                  React.createElement('div', { key: 'divider', style: { borderTop: '1px solid rgba(148,163,184,0.25)', margin: '8px 0' } }),
                  React.createElement('div', { key: 'sl', style: { fontSize: 11, color: '#64748b', marginBottom: 4 } }, '处理器技能内容:'),
                  React.createElement('pre', {
                    style: {
                      margin: 0, whiteSpace: 'pre-wrap', wordBreak: 'break-word',
                      fontSize: 11, lineHeight: 1.5, color: '#334155',
                      background: 'rgba(148,163,184,0.08)', borderRadius: 6,
                      padding: '8px 10px', maxHeight: 300, overflow: 'auto',
                    },
                  }, skillText),
                ])
              }
            }

            const bottomEl = React.createElement('div', {
              key: 'bottom',
              style: { flex: 1, overflow: 'auto', padding: '10px 16px 16px' },
            }, [
              React.createElement('div', {
                key: 'bh',
                style: { fontSize: 12, color: '#64748b', marginBottom: 6 },
              }, detailTitle),
              detailContent || React.createElement('div', {
                key: 'hint',
                style: { fontSize: 12, color: '#9ca3af' },
              }, '点击上方节点查看其配置与技能内容'),
            ])

            const dagEl = React.createElement(DagCanvas, {
              key: 'dag',
              stage: stateData.stage,
              gateResult: stateData.gateResult || null,
              tasks,
              selectedId,
              onSelect,
              workflowName: stateData.workflow,
              retries: stateData.retries || 0,
              error: stateData.error || null,
            })

            return React.createElement('div', {
              style: {
                display: 'flex', flexDirection: 'column', height: '100%',
                minHeight: 420, fontFamily: 'inherit', fontSize: 13,
              },
            }, [dagEl, bottomEl])
          }

          return React.createElement(WorkflowView)
        },
      )
    })
  },
}
